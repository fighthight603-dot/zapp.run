import 'package:flutter/material.dart';

void main() => runApp(TikTokClone());

class TikTokClone extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: Scaffold(
        backgroundColor: Colors.black,
        body: VideoFeed(),
      ),
    );
  }
}

class VideoFeed extends StatelessWidget {
  // A sample list of colors to represent videos for this test
  final List<Color> videoColors = [Colors.red, Colors.blue, Colors.green, Colors.orange];

  @override
  Widget build(BuildContext context) {
    return PageView.builder(
      scrollDirection: Axis.vertical, // The magic "TikTok" swipe
      itemCount: videoColors.length,
      itemBuilder: (context, index) {
        return Stack(
          children: [
            // This represents the Video Area
            Container(
              color: videoColors[index],
              child: Center(
                child: Column(
                  mainAxisAlignment: MainAxisAlignment.center,
                  children: [
                    Icon(Icons.play_circle_fill, size: 80, color: Colors.white54),
                    Text("Video $index", style: TextStyle(color: Colors.white, fontSize: 24)),
                  ],
                ),
              ),
            ),
            // The Overlay UI (Like, Comment, Profile)
            Positioned(
              right: 15,
              bottom: 100,
              child: Column(
                children: [
                  CircleAvatar(backgroundColor: Colors.white, child: Icon(Icons.person)),
                  SizedBox(height: 20),
                  Icon(Icons.favorite, color: Colors.white, size: 35),
                  Text("1.2k", style: TextStyle(color: Colors.white)),
                  SizedBox(height: 20),
                  Icon(Icons.comment, color: Colors.white, size: 35),
                  Text("45", style: TextStyle(color: Colors.white)),
                  SizedBox(height: 20),
                  Icon(Icons.share, color: Colors.white, size: 35),
                  Text("Share", style: TextStyle(color: Colors.white)),
                ],
              ),
            ),
            // The Caption Area
            Positioned(
              left: 15,
              bottom: 30,
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text("@nepal_creator", style: TextStyle(color: Colors.white, fontWeight: FontWeight.bold)),
                  SizedBox(height: 10),
                  Text("This is my first video on my new app! #nepal #tech", style: TextStyle(color: Colors.white)),
                ],
              ),
            ),
          ],
        );
      },
    );
  }
}
